This section presents the mathematical formulation of the model and its explanation.  
In the following, \textbf{endogenous variables} begin with uppercase letters,
while \textbf{parameters} begin with lowercase letters or use Greek notation.
Lowercase parameters include calibrated coefficients and exogenous or predetermined inputs.

\subsection{Demand Block}
This block describes the equations related to demand determination.  
In the demand block, food demand is determined by factors such as prices, population, and GDP.

\paragraph{Total Demand}
% The total domestic demand (\( QD_{c,cty} \)) is the sum of all types of demand, including household consumption, feed, processing, biofuel, seed, tourism, food loss, food waste, and other demand.
\begin{align}
  QD_{c,cty}
  &= QDFood_{c,cty} + QDFed_{c,cty} + QDInt_{c,cty} + QDBio_{c,cty} \nonumber\\
  &\quad + QDSeed_{c,cty} + QDTour_{c,cty} + QDLoss_{c,cty}
    + QDWaste_{c,cty} + QDOth_{c,cty}
  \tag{QDEq}\label{eq:QDEq}
\end{align}

\paragraph{Consumer Price Index}
\begin{align}
  PCIdx_{c,cty}
  &= \frac{(1-rPcSub_{c,cty})\cdot PC_{c,cty}}
           {(1-rPcSub0_{c,cty})\cdot pc0_{c,cty}}
  \tag{PCIdxEq}\label{eq:PCIdxEq}
\end{align}

\paragraph{Household Consumption Demand}
\begin{align}
  QDFood_{c,cty}
  &= qdFood0_{c,cty}
     \cdot
     \left(\frac{gdpPc_{cty}}{gdpPc0_{cty}}\right)^{elaInc_{c,cty}}
     \nonumber\\
  &\quad \cdot
     \prod_{c2}\!
     PCIdx_{c2,cty}^{elaP_{c,c2,cty}}
     \cdot
     \frac{pop_{cty}}{pop0_{cty}}
  \tag{QDFoodEq}\label{eq:QDFoodEq}
\end{align}

\paragraph{Feed Demand}
\begin{align}
  QDFed_{c\_fed,cty}
  &= \sum_{bas}\sum_{a\_liv} QFeed_{a\_liv,c\_fed,bas,cty}
     \nonumber\\[-0.3em]
  &\quad + \sum_{a\_fsh} QSFsh_{a\_fsh,cty,\text{"aqu"}}\cdot FeedCoefCty_{a\_fsh,c\_fed,cty}
  \tag{QDFedEq}\label{eq:QDFedEq}
\end{align}

\paragraph{Intermediate Input Demand for Processing}
\begin{align}
  QDInt_{c\_int,cty}
  &= \sum_{a\_pro} QInt_{a\_pro,c\_int,cty}
  \tag{QDIntEq}\label{eq:QDIntEq}
\end{align}

\paragraph{Biofuel Demand}
\begin{align}
  QDBio_{c,cty}
  &= qdBio0_{c,cty}
     \cdot
     \prod_{c2}\!
     PCIdx_{c2,cty}^{elaP_{c,c2,cty}}
  \tag{QDBioEq}\label{eq:QDBioEq}
\end{align}

\paragraph{Seed Demand}
\begin{align}
  QDSeed_{c,cty}
  &= qdSeed0_{c,cty}
     \cdot\left(\frac{gdpPc_{cty}}{gdpPc0_{cty}}\right)^{elaInc_{c,cty}}
     \cdot\frac{pop_{cty}}{pop0_{cty}}
     \nonumber\\
  &\quad\cdot\prod_{c2}PCIdx_{c2,cty}^{elaP_{c,c2,cty}}
  \tag{QDSeedEq}\label{eq:QDSeedEq}
\end{align}

\paragraph{Tourist Consumption Demand}
\begin{align}
  QDTour_{c,cty}
  &= qdTour0_{c,cty}
     \cdot\left(\frac{gdpPc_{cty}}{gdpPc0_{cty}}\right)^{elaInc_{c,cty}}
     \cdot\frac{pop_{cty}}{pop0_{cty}}
     \nonumber\\
  &\quad\cdot\prod_{c2}PCIdx_{c2,cty}^{elaP_{c,c2,cty}}
  \tag{QDTourEq}\label{eq:QDTourEq}
\end{align}

\paragraph{Loss and Waste}
\begin{align}
  QDLoss_{c,cty}
  &= QD_{c,cty} \cdot rLoss_{c,cty}
  \tag{QDLossEq}\label{eq:QDLossEq}\\[.3em]
  QDWaste_{c,cty}
  &= QDFood_{c,cty} \cdot rWaste_{c,cty}
  \tag{QDWasteEq}\label{eq:QDWasteEq}
\end{align}

\paragraph{Other Demand}
\begin{align}
  QDOth_{c,cty}
  &= qdOth0_{c,cty}
     \cdot\left(\frac{gdpPc_{cty}}{gdpPc0_{cty}}\right)^{elaInc_{c,cty}}
     \cdot\frac{pop_{cty}}{pop0_{cty}}
     \nonumber\\
  &\quad\cdot\prod_{c2}PCIdx_{c2,cty}^{elaP_{c,c2,cty}}
  \tag{QDOthEq}\label{eq:QDOthEq}
\end{align}


\subsection{Production Block}

The production block describes the equations determining the production quantities of each food commodity, the input of intermediate goods and production factors, and the producer prices.  
The production functions are represented by Leontief-type functions, where each good is produced through the combination of specific input factors.

\subsubsection{Annual Driver Calculations}

Crop yields, livestock system shares, and feed input coefficients are calculated before each annual or sub-step equilibrium solve. They are therefore time-varying parameters rather than MCP equilibrium variables:

\begin{align*}
  yld_{a\_cro,bas,cty,wat}
  &= yld0_{a\_cro,bas,cty,wat}
     \cdot yldTech_{a\_cro,bas,cty,wat}
     \\
  &\quad
     \cdot yldChg_{a\_cro,bas,cty,wat}
     \cdot yldPlum_{a\_cro,bas,cty},\\
  shrSys_{a\_liv\_rum,bas,cty,cz,\text{"LG"}}
  &= \frac{1}{1+\delta^{sys}_{a\_liv\_rum,bas,cty,cz,\text{"LG"}}
     \cdot gdpPc_{cty}^{-\rho^{sys}_{a\_liv\_rum,\text{"LG"}}}},\\
  shrSys_{a\_liv\_rum,bas,cty,cz,\text{"MX"}}
  &= 1-shrSys_{a\_liv\_rum,bas,cty,cz,\text{"LG"}},\\
  feedCoef_{a\_liv,c\_fed,bas,cty,cz,sys}
  &= feedReq_{a\_liv,bas,cty,cz,sys}
     \cdot shrFeedConc_{a\_liv,bas,cty,cz,sys}
     \\
  &\quad
     \cdot shrFeedGrain_{a\_liv,c\_fed,bas,cty,cz,sys}.
\end{align*}

The optional \(yldPlum\) term is one when AIM-PLUM feedback is not used. The LG/MX split applies to ruminants; non-ruminants use the aggregate system.

\paragraph{Domestic Production}
\begin{align}
  QS_{a\_bas,cty}
  &= \sum_{bas} QSBas_{a\_bas,bas,cty}
  \tag{QSAggEq}\label{eq:QSAggEq}
\end{align}

\subsubsection{Crop Production}

Primary crops are differentiated by production technologies (irrigated and rainfed).
Yield improvements due to technological progress and climate impacts are considered for crops. Crop yields also respond to land rents.

\paragraph{Crop Production by Technology}
\begin{align}
  QSWat_{a\_cro,bas,cty,wat}
  &= QSBas_{a\_cro,bas,cty} \cdot shrWat_{a\_cro,bas,cty,wat}
  \tag{QSWatEq}\label{eq:QSWatEq}
\end{align}

\paragraph{Producer Price of Crops}
\begin{multline}
  PP_{a\_cro,cty}\cdot(1+rPpSub_{a\_cro,cty}-rPpTax_{a\_cro,cty}) \\
  = \Bigl(
      PL_{a\_cro,bas,cty}
      \cdot
      \left(\frac{PL_{a\_cro,bas,cty}}{pl0_{a\_cro,bas,cty}}\right)^{-elaYldPL}
      \cdot
      \sum_{wat}
      \frac{
        shrWat_{a\_cro,bas,cty,wat}
      }{
        yld_{a\_cro,bas,cty,wat}
        \cdot
        croInt_{bas,cty,a\_cro,wat}
      }
      \\[-0.3em]
  \quad
      +\; COBas_{a\_cro,bas,cty}
     \Bigl)
     \,/\, (1 + MuBasMin_{a\_cro,bas,cty})
  \tag{CroZeroProfitEq}\label{eq:CroZeroProfitEq}
\end{multline}

\paragraph{Land Input for Crop Production}
\begin{multline}
  QL_{a\_cro,bas,cty}
  = QSBas_{a\_cro,bas,cty}
     \cdot
     \left(\frac{PL_{a\_cro,bas,cty}}{pl0_{a\_cro,bas,cty}}\right)^{-elaYldPL}
     \\[-0.3em]
     \cdot
     \sum_{wat}
     \frac{
       shrWat_{a\_cro,bas,cty,wat}
     }{
       yld_{a\_cro,bas,cty,wat}
       \cdot
       croInt_{bas,cty,a\_cro,wat}
     }
  \tag{CroLandEq}\label{eq:CroLandEq}
\end{multline}

\paragraph{Other Production Factor Price (Crops)}
\begin{align}
  COBas_{a\_cro,bas,cty}
  &= coBas0_{a\_cro,bas,cty}
     \cdot
     \left(
       \frac{QL_{a\_cro,bas,cty}}
            {ql0_{a\_cro,bas,cty}}
     \right)^{elaCO_{a\_cro,cty}}
  \tag{COCroEq}\label{eq:COCroEq}
\end{align}


\subsubsection{Livestock Production}

The livestock production block describes equations for determining livestock output, production systems, feed requirements, and associated prices.  
Livestock products are distinguished by climate zones and production systems (pasture-based or mixed).  
Feed use and production efficiency depend on feed composition, feed requirements, and animal productivity.

\paragraph{Livestock Production by System}
\begin{align}
  QSSys_{a\_liv,bas,cty,cz,sys}
  &= QSBas_{a\_liv,bas,cty}
     \cdot shrCz_{a\_liv,bas,cty,cz}
     \cdot shrSys_{a\_liv,bas,cty,cz,sys}
  \tag{QSSysEq}\label{eq:QSSysEq}
\end{align}

\paragraph{Producer Price of Livestock}
\begin{multline}
  PP_{a\_liv,cty}\cdot(1+rPpSub_{a\_liv,cty}-rPpTax_{a\_liv,cty}) \\
  = \Biggl(\frac{PL_{a\_liv,bas,cty}}{YLiv_{a\_liv,bas,cty}}
       + COBas_{a\_liv,bas,cty}
       + feedCostRes_{a\_liv,bas,cty}
       \\[-0.3em]
  \quad + \sum_{c\_fed}pFeed0_{a\_liv,c\_fed,cty}
       \cdot\frac{PC_{c\_fed,cty}}{pc0_{c\_fed,cty}}
       \\[-0.3em]
  \quad \cdot \sum_{cz,sys}\Bigl(
       shrCz_{a\_liv,bas,cty,cz}\cdot
       shrSys_{a\_liv,bas,cty,cz,sys}\cdot
       feedCoef_{a\_liv,c\_fed,bas,cty,cz,sys}\Bigr)\Biggr)
       \\[-0.3em]
  \quad / (1+MuBasMin_{a\_liv,bas,cty})
  \tag{LivZeroProfitEq}\label{eq:LivZeroProfitEq}
\end{multline}

\paragraph{Grass Input Calculation}
\begin{align}
  QGrass_{a\_liv,bas,cty}
  &=
     QSBas_{a\_liv,bas,cty}
     \nonumber\\[-0.3em]
  &\quad
     \cdot
     \sum_{cz,sys}
     \Bigl(
       shrCz_{a\_liv,bas,cty,cz}
       \cdot
       shrSys_{a\_liv,bas,cty,cz,sys}
       \nonumber\\[-0.3em]
  &\quad
       \cdot
       feedReq_{a\_liv,bas,cty,cz,sys}
       \cdot
       (1 - shrFeedConc_{a\_liv,bas,cty,cz,sys})
     \Bigr)
  \tag{QGrassEq}\label{eq:QGrassEq}
\end{align}

\paragraph{Available Grassland Area}
\begin{align}
  QGrass_{a\_liv,bas,cty}
  &=
     QL_{a\_liv,bas,cty}
     \cdot
     yGrass_{bas,cty}
  \tag{GrassBalEq}\label{eq:GrassBalEq}
\end{align}


\paragraph{Livestock Yield}
\begin{align}
  YLiv_{a\_liv,bas,cty}
  &= \frac{
       QSBas_{a\_liv,bas,cty}
     }{
       QL_{a\_liv,bas,cty}
     }
  \tag{YLivEq}\label{eq:YLivEq}
\end{align}

\paragraph{Feed Input Calculation}
\begin{align}
  QFeed_{a\_liv,c\_fed,bas,cty}
  &= QSBas_{a\_liv,bas,cty}
     \nonumber\\[-0.3em]
  &\quad
     \cdot
     \sum_{cz,sys}
     \bigl(
       shrCz_{a\_liv,bas,cty,cz}
       \cdot shrSys_{a\_liv,bas,cty,cz,sys}
       \nonumber\\[-0.3em]
  &\quad
       \cdot feedCoef_{a\_liv,c\_fed,bas,cty,cz,sys}
     \bigr)
  \tag{QFeedEq}\label{eq:QFeedEq}
\end{align}

\paragraph{Other Production Factor Price (Livestock)}
\begin{align}
  COBas_{a\_liv,bas,cty}
  &= coBas0_{a\_liv,bas,cty}
     \cdot\left(\frac{QL_{a\_liv,bas,cty}}{ql0_{a\_liv,bas,cty}}\right)^{elaCO_{a\_liv,cty}}
  \tag{COLivEq}\label{eq:COLivEq}
\end{align}
The land-cost term applies only to grazing livestock. For livestock without land input, the cost ratio in Eq.~\eqref{eq:COLivEq} is \(QSBas/qsBas0\).

\paragraph{Livestock Production Quantity}
\begin{align}
  QSBas_{a\_liv,bas,cty}
  &= animalWt_{a\_liv,bas,cty}
     \cdot
     (
       NAnimal_{a\_liv,bas,cty}
       - nAnimalRes_{a\_liv,bas,cty}
     )
     \nonumber\\[-0.3em]
  &\quad
     \cdot
     rAnimalProd_{a\_liv,cty}
  \tag{NAnimalEq}\label{eq:NAnimalEq}
\end{align}

\subsubsection{Fishery Production}

Fishery products are produced at the country level by capture fisheries and aquaculture.
Capture production is determined by its base-year level and a price elasticity, and aquaculture is produced using feed and other production factors.
Reduction fish is supplied only by capture fisheries and is used as the raw material of fishmeal and fish oil.

\paragraph{Fishery Production by System}
\begin{align}
  QS_{a\_fsh,cty}
  &= \sum_{fsys} QSFsh_{a\_fsh,cty,fsys}
  \tag{QSFshEq}\label{eq:QSFshEq}
\end{align}

\paragraph{Capture Fishery Production}
\begin{align}
  QSFsh_{a\_fsh,cty,\text{"cap"}}
  &= qsFsh0_{a\_fsh,cty,\text{"cap"}}
     \cdot\left(\frac{PP_{a\_fsh,cty}}{pp0_{a\_fsh,cty}}\right)^{elaCap}
  \tag{FshCapEq}\label{eq:FshCapEq}
\end{align}

\paragraph{Producer Price of Aquaculture}
\begin{multline}
  PP_{a\_fsh,cty}\cdot(1+rPpSub_{a\_fsh,cty}-rPpTax_{a\_fsh,cty}) \\
  = \Biggl(COCty_{a\_fsh,cty}
     \\[-0.3em]
  \quad + \sum_{c\_fed}pFeed0_{a\_fsh,c\_fed,cty}
       \cdot\frac{PC_{c\_fed,cty}}{pc0_{c\_fed,cty}}\cdot
       FeedCoefCty_{a\_fsh,c\_fed,cty}\Biggr)
     \\[-0.3em]
  \quad / (1+MuFshMin_{a\_fsh,cty})
  \tag{FshZeroProfitEq}\label{eq:FshZeroProfitEq}
\end{multline}
Aquaculture whose base-year feed cost exceeds its producer revenue keeps its base-year output.

\paragraph{Other Production Factor Price (Aquaculture)}
\begin{align}
  COCty_{a\_fsh,cty}
  &= coCty0_{a\_fsh,cty}
     \cdot\left(\frac{QSFsh_{a\_fsh,cty,\text{"aqu"}}}{qsFsh0_{a\_fsh,cty,\text{"aqu"}}}\right)^{elaCO_{a\_fsh,cty}}
  \tag{COFshEq}\label{eq:COFshEq}
\end{align}

\paragraph{Aquaculture Feed Input}
When fishmeal or fish oil becomes more expensive relative to oilseed cake, aquaculture replaces it with the same weight of oilseed cake. The other feed inputs keep their base-year coefficients.
\begin{align}
  FeedCoefCty_{a\_fsh,c\_fed\_fsh,cty}
  &= feedCoefCty0_{a\_fsh,c\_fed\_fsh,cty}
     \nonumber\\[-0.3em]
  &\quad \cdot \left(\frac{PC_{c\_fed\_fsh,cty}/pc0_{c\_fed\_fsh,cty}}
                {PC_{\text{"cak"},cty}/pc0_{\text{"cak"},cty}}\right)^{-elaFeedFsh}
  \tag{FeedCoefFshEq}\label{eq:FeedCoefFshEq}\\[0.4em]
  FeedCoefCty_{a\_fsh,\text{"cak"},cty}
  &= feedCoefCty0_{a\_fsh,\text{"cak"},cty}
     \nonumber\\[-0.3em]
  &\quad + \sum_{c\_fed\_fsh} feedCoefCty0_{a\_fsh,c\_fed\_fsh,cty}
     \nonumber\\[-0.3em]
  &\quad - \sum_{c\_fed\_fsh} FeedCoefCty_{a\_fsh,c\_fed\_fsh,cty}
  \tag{FeedCoefCakEq}\label{eq:FeedCoefCakEq}
\end{align}

\subsubsection{Processed Goods Production}

This section describes the production equations for processed agricultural products.  
Processed products are produced using intermediate inputs and other production factors.

\paragraph{Producer Price of Processed Products}
\begin{multline}
  PP_{a\_pro,cty}\cdot(1+rPpSub_{a\_pro,cty}-rPpTax_{a\_pro,cty}) \\
  = COCty_{a\_pro,cty}
     \\[-0.3em]
  \quad + \sum_{c\_int}pInt0_{a\_pro,c\_int,cty}
       \cdot\frac{PC_{c\_int,cty}}{pc0_{c\_int,cty}}\cdot
       intCoef_{a\_pro,c\_int,cty}
  \tag{ProZeroProfitEq}\label{eq:ProZeroProfitEq}
\end{multline}

\paragraph{Intermediate Input Demand for Processed Products}
\begin{align}
  QInt_{a\_pro,c\_int,cty}
  &= QS_{a\_pro,cty}
     \cdot
     intCoef_{a\_pro,c\_int,cty}
  \tag{QIntEq}\label{eq:QIntEq}
\end{align}

\paragraph{Other Production Factor Price (Processed Products)}
\begin{align}
  COCty_{a\_pro,cty}
  &= coCty0_{a\_pro,cty}
     \cdot
     \left(
       \frac{
         QS_{a\_pro,cty}
       }{
         qs0_{a\_pro,cty}
       }
     \right)^{elaCO_{a\_pro,cty}}
  \tag{COProEq}\label{eq:COProEq}
\end{align}

\subsubsection{Energy Crop Production}
Energy crop production is determined from regional totals provided exogenously by the AIM-Hub model~\cite{fujimoriSSP3AIMImplementation2017}.


\paragraph{Regional Energy Crop Supply Balance}
\begin{align}
  qdEne_{reg}
  &= \sum_{bas,cty \in reg}
     QSBas_{\text{"ecr"},bas,cty}
  \tag{EneBalEq}\label{eq:EneBalEq}
\end{align}

\paragraph{Land Input for Energy Crops}
\begin{align}
  QL_{\text{"ecr"},bas,cty}
  &= \frac{
       QSBas_{\text{"ecr"},bas,cty}
     }{
       yld_{\text{"ecr"},bas,cty,\text{"Rainfed"}}
     }
  \tag{EneLandEq}\label{eq:EneLandEq}
\end{align}

\paragraph{Land Price for Energy Crops}
\begin{align}
  PL_{\text{"ecr"},bas,cty}
  &= PLEne_{reg}
  \tag{PLEneEq}\label{eq:PLEneEq}
\end{align}


\subsection{Price Block}

This subsection describes the equations in the price block.  
The price block represents the relationships among various prices in the model.  
Prices are linked through fixed coefficients so that the price used in the demand and supply functions is consistently determined.

\paragraph{Consumer and Producer Prices}
For non-traded goods,
\begin{align}
  PC_{c,cty}
  &= \frac{(1+mmj_{c,cty})\cdot PP_{c,cty}
       + CAbt_{c,cty}+CEmis_{c,cty}}
       {1+MuNtrdMin_{c,cty}}
  \tag{PCNtrdEq}\label{eq:PCNtrdEq}
\end{align}
For traded goods, the domestic-good price satisfies
\begin{align}
  PB_{c,cty}\cdot(1+pbRes_{c,cty})
  &= (1+mmj_{c,cty})\cdot PP_{c,cty}
     \nonumber\\[-0.3em]
  &\quad + CAbt_{c,cty}+CEmis_{c,cty}
  \tag{PBEq}\label{eq:PBEq}
\end{align}

\paragraph{Consumer and Border Prices}
Under the price-parity trade formulation, the consumer price is weighted by base-year domestic supply and imports:
\begin{align}
  PC_{c,cty}
  &= PB_{c,cty}\cdot(1+pbRes_{c,cty})
     \nonumber\\[-0.3em]
  &\quad + \Bigl[PM_{c,cty}-PB_{c,cty}\cdot(1+pbRes_{c,cty})\Bigr]
       \cdot\frac{qm0_{c,cty}}
       {qs0_{c,cty}-qe0_{c,cty}+qm0_{c,cty}}
  \tag{PCTrdEq}\label{eq:PCTrdEq}
\end{align}
The import term is zero where base-year imports are zero.

\paragraph{Import Price}
\begin{align}
  PM_{c,cty}
  &= (PW_c+frtM_{c,cty})\cdot exr_{cty}
     \nonumber\\[-0.3em]
  &\quad \cdot (1+tm_{c,cty}+MuImpMin_{c,cty})
     \cdot (1+pmRes_{c,cty})
  \tag{PMEq}\label{eq:PMEq}
\end{align}

\paragraph{Export Price}
For the price-parity trade formulation,
\begin{align}
  PE_{c,cty}
  &= \frac{(PW_c-frtE_{c,cty})\cdot
       (1+peRes_{c,cty})\cdot exr_{cty}}
       {1+te_{c,cty}+MuExpMax_{c,cty}}
  \tag{PEEq}\label{eq:PEEq}
\end{align}

\subsection{Land Use Block}

The land-use block allocates available land among different uses through a Logit function. Each nest uses a common exponent.
Non-pasture vegetation is allocated among cropland, forest, other natural land, and afforested land. Protected areas are imposed as lower bounds for the corresponding land classes, and the agro-ecological suitability constraint applies to cropland.  
Urban land is assigned exogenously. In the default configuration, the LUH3-based base-year urban area is held constant over future years; as an optional extension, SSP-specific urban projections from the SEDAC urban dataset can be used to update urban land over time.

\paragraph{Vegetation Area}
\begin{align}
  QLVeg_{bas,cty}
  &= 
     qlTot_{bas,cty}
     - qlOth_{bas,cty}
     - qlUrb_{bas,cty}
  \tag{QLVegBalEq}\label{eq:QLVegBalEq}
\end{align}

\paragraph{Pasture Area Allocation}
\begin{align}
  QLPas_{bas,cty}
  &=
    QLVeg_{bas,cty}
    \nonumber\\[-0.3em]
  &\quad\cdot
    \frac{
      \delta^{\text{pas}}_{bas,cty}
      \cdot
      PLPas_{bas,cty}^{\,\rho^{\text{pas}}_{bas,cty}}
    }{
      \delta^{\text{pas}}_{bas,cty}
      \cdot
      PLPas_{bas,cty}^{\,\rho^{\text{pas}}_{bas,cty}}
      +
      \delta^{\text{npas}}_{bas,cty}
      \cdot
      PLNpas_{bas,cty}^{\,\rho^{\text{npas}}_{bas,cty}}
    }
  \tag{PasLogitEq}\label{eq:PasLogitEq}
\end{align}

\paragraph{Grazed and Ungrazed Pasture Allocation}
\begin{align}
  QLPasGrz_{bas,cty}
  &=
    QLPas_{bas,cty}
    \cdot
    \frac{
      \delta^{\text{pasgrz}}_{bas,cty}
      \cdot
      PLPasGrz_{bas,cty}^{\,\rho^{\text{pasgrz}}_{bas,cty}}
    }{
      \delta^{\text{pasgrz}}_{bas,cty}
      \cdot
      PLPasGrz_{bas,cty}^{\,\rho^{\text{pasgrz}}_{bas,cty}}
      +
      \delta^{\text{pasung}}_{bas,cty}
      \cdot
      PLPasUng_{bas,cty}^{\,\rho^{\text{pasung}}_{bas,cty}}
    }
  \tag{PasGrzLogitEq}\label{eq:PasGrzLogitEq}
\end{align}

\begin{align}
  QLPasUng_{bas,cty}
  &=
    QLPas_{bas,cty}
    \cdot
    \frac{
      \delta^{\text{pasung}}_{bas,cty}
      \cdot
      PLPasUng_{bas,cty}^{\,\rho^{\text{pasung}}_{bas,cty}}
    }{
      \delta^{\text{pasgrz}}_{bas,cty}
      \cdot
      PLPasGrz_{bas,cty}^{\,\rho^{\text{pasgrz}}_{bas,cty}}
      +
      \delta^{\text{pasung}}_{bas,cty}
      \cdot
      PLPasUng_{bas,cty}^{\,\rho^{\text{pasung}}_{bas,cty}}
    }

\subsection{Market Block}

This subsection describes the equations in the market block, where consistency between food demand and supply is ensured.  
Trade of food commodities is represented as a complementarity problem.  
When the import price is higher than the border price, the import quantity becomes zero.  
Once the import price decreases to equal the border price, imports begin.  
Similarly, when the border price is higher than the export price, exports become zero, and once the export price rises to match the border price, exports take place.

\paragraph{Domestic Trade Balance}
\begin{align}
  NT_{c,cty}
  &= QS_{c,cty} - QD_{c,cty} - qStock_{c,cty}
  \tag{ComBalEq}\label{eq:ComBalEq}\\[0.6em]
  NT_{c,cty}
  &= QE_{c,cty} - QM_{c,cty}
  \tag{NTEq}\label{eq:NTEq}
\end{align}

\paragraph{International Trade Balance}
\begin{align}
  \sum_{cty} NT_{c,cty} - qRes_{c} = 0
  \tag{WldBalEq}\label{eq:WldBalEq}
\end{align}

\paragraph{Import and Export Conditions}
\begin{align}
  PM_{c,cty} &\ge PB_{c,cty}
  \quad \perp \quad
  QM_{c,cty} \ge 0
  \tag{ImpParityEq}\label{eq:ImpParityEq}\\[0.4em]
  PB_{c,cty} &\ge PE_{c,cty}
  \quad \perp \quad
  QE_{c,cty} \ge 0
  \tag{ExpParityEq}\label{eq:ExpParityEq}
\end{align}



\subsection{GHG Emission Block}

This subsection describes the equations in the GHG emission block.  
GHG emissions are calculated based on the production quantities of each food commodity.  
When carbon prices are considered, the amount of mitigation technology adoption and its cost are computed from the MAC (Marginal Abatement Cost) curve.  
Emissions from the land-use sector are calculated according to each land area.

\paragraph{GHG Emissions}
\begin{align}
  GHG_{a,src,gas,cty}
  &= QS_{a,cty}
     \cdot
     ef_{a,src,gas,cty}
     \cdot
     (1 - RRed_{a,src,gas,cty})
  \tag{GHGAgEq}\label{eq:GHGAgEq}
\end{align}

\paragraph{Land-Use Emissions}
\begin{multline}
  GHGLU_{src,gas,cty}
  =
  \sum_{bas}\Bigl(
    efLUCro_{src,gas,cty}\cdot QLCro_{bas,cty}
    \\[-0.3em]
    + efLUFrs_{src,gas,cty}\cdot QLFrs_{bas,cty}
    \\[-0.3em]
    - carbonStock_{bas,cty}\cdot\frac{44}{12}
      \cdot \min\!\Bigl\{
        0,\,
        QLFrs_{bas,cty}-qlFrsPrev_{bas,cty}
      \Bigr\}
    \\[-0.3em]
    + efLUPas_{src,gas,cty}\cdot QLPas_{bas,cty}
    + efLUNat_{src,gas,cty}\cdot QLNat_{bas,cty}
  \Bigr)
  \tag{GHGLUEq}\label{eq:GHGLUEq}
\end{multline}
\begin{align}
  GHGLU_{\text{"afr"},\text{"CO\textsubscript{2}"},cty}
  &=
  - \sum_{bas}\sum_{age}
      qlAfrAge_{bas,cty,age}
      \cdot carbonSeq_{bas,cty,age}
  \tag{AfrGHGEq}\label{eq:AfrGHGEq}
\end{align}

\paragraph{Emission Reduction Rate}
\begin{align}
  RRed_{a,src,gas,cty}
  &= mrp_{src,gas,cty}
     \cdot
     \Bigl(
       1 -
       \bigl[
         1 / (1 + pCar_{cty})
       \bigr]^{\sigma_{src,gas,cty}}
     \Bigr)
  \tag{RRedEq}\label{eq:RRedEq}
\end{align}

\paragraph{Abatement Cost}
\begin{align}
  CAbt_{a,cty}
  &= \sum_{src,gas}
     ef_{a,src,gas,cty}
     \cdot
     \Biggl[
       \frac{\sigma_{src,gas,cty}}
            {(\sigma_{src,gas,cty}-1)}
       \cdot
       mrp_{src,gas,cty}
       \nonumber\\[-0.3em]
  &\quad
       \cdot
       \Bigl(
         1 -
         \bigl(
           1 -
           RRed_{a,src,gas,cty} /
           mrp_{src,gas,cty}
         \bigr)^{(1 - 1/\sigma_{src,gas,cty})}
       \Bigr)
       - RRed_{a,src,gas,cty}
     \Biggr]
  \tag{CAbtEq}\label{eq:CAbtEq}
\end{align}

\paragraph{Carbon Cost on Remaining Emissions}
The factor \(shrCEmis\) limits the charge relative to the previous consumer price.
\begin{align}
  CEmis_{a,cty}
  &= shrCEmis_{a,cty}\cdot\sum_{src,gas}
     pCar_{cty}
     \cdot
     ef_{a,src,gas,cty}
     \cdot
     (1 - RRed_{a,src,gas,cty})
  \tag{CEmisEq}\label{eq:CEmisEq}
\end{align}


\subsection{Complementarity Conditions}

This subsection describes the complementarity conditions introduced to avoid numerical errors in the Logit function when production quantities in specific production units become zero.  
If the production of a unit becomes zero, its land-use area would also become zero, leading to instability.  
Therefore, a minimum positive value is imposed to ensure that production quantities do not fall to zero.

\paragraph{Minimum Production Share for Trade}
\begin{align}
  \frac{QS_{c,cty}}{QM_{c,cty}}
  &\ge 0.01
  \quad \perp \quad
  MuImpMin_{c,cty} \ge 0
  \tag{ImpMinEq}\label{eq:ImpMinEq}
\end{align}

\paragraph{Minimum Production Share for Supply}
\begin{align}
  \frac{QSBas_{a,bas,cty}}{QS_{a,cty}}
  &\ge 0.01
  \quad \perp \quad
  MuBasMin_{a,bas,cty} \ge 0
  \tag{QSBasMinEq}\label{eq:QSBasMinEq}
\end{align}

\paragraph{Minimum Aquaculture Production}
\begin{align}
  \frac{QSFsh_{a\_fsh,cty,\text{"aqu"}}}{qsFsh0_{a\_fsh,cty,\text{"aqu"}}}
  &\ge 0.01
  \quad \perp \quad
  MuFshMin_{a\_fsh,cty} \ge 0
  \tag{QSFshMinEq}\label{eq:QSFshMinEq}
\end{align}
