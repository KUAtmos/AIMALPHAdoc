This section presents the mathematical formulation of the model and its explanation.  
In the following, \textbf{endogenous variables} begin with uppercase letters,
while \textbf{parameters} begin with lowercase letters or use Greek notation.
An overline identifies an exogenous or predetermined value.

\subsection{Demand Block}
This block describes the equations related to demand determination.  
In the demand block, food demand is determined by factors such as prices, population, and GDP.

\paragraph{Total Demand}
% The total domestic demand (\( QD_{c,cty} \)) is the sum of all types of demand, including household consumption, feed, processing, biofuel, seed, tourism, food loss, food waste, and other demand.
\begin{align}
  QD_{c,cty}
  &= QDFood_{c,cty} + QDFed_{c,cty} + QDInt_{c,cty} + QDBio_{c,cty} \nonumber\\
  &\quad + QDSeed_{c,cty} + QDTour_{c,cty} + QDLoss_{c,cty}
    + QDWaste_{c,cty} + QDOth_{c,cty}
  \tag{QDEq}\label{eq:QDEq}
\end{align}

\paragraph{Household Consumption Demand}
\begin{align}
  QDFood_{c,cty}
  &= \overline{qdFood0}_{c,cty}
     \cdot
     \left(\frac{\overline{gdpPc}_{cty}}{\overline{gdpPc0}_{cty}}\right)^{elaInc_{c,cty}}
     \nonumber\\
  &\quad \cdot
     \prod_{c2}\!
     \left(
       \frac{\bigl(1-\overline{cse}_{c2,cty}\bigr)\cdot PC_{c2,cty}}
            {\bigl(1-\overline{cse0}_{c2,cty}\bigr)\cdot \overline{pc0}_{c2,cty}}
     \right)^{elaP_{c,c2,cty}}
     \cdot
     \frac{\overline{pop}_{cty}}{\overline{pop0}_{cty}}
  \tag{QDFoodEq}\label{eq:QDFoodEq}
\end{align}

\paragraph{Feed Demand}
\begin{align}
  QDFed_{c\_fed,cty}
  &= \sum_{bas}\sum_{a\_liv} QFeed_{a\_liv,c\_fed,cty}
  \tag{QDFedEq}\label{eq:QDFedEq}
\end{align}

\paragraph{Intermediate Input Demand for Processing}
\begin{align}
  QDInt_{c\_int,cty}
  &= \sum_{a\_pro} QInt_{a\_pro,c\_int,bas,cty}
  \tag{QDIntEq}\label{eq:QDIntEq}
\end{align}

\paragraph{Biofuel Demand}
\begin{align}
  QDBio_{c,cty}
  &= \overline{qdBio0}_{c,cty}
     \cdot
     \prod_{c2}\!
     \left(
       \frac{\bigl(1-\overline{cse}_{c2,cty}\bigr)\cdot PC_{c2,cty}}
            {\bigl(1-\overline{cse0}_{c2,cty}\bigr)\cdot \overline{pc0}_{c2,cty}}
     \right)^{elaP_{c,c2,cty}}
  \tag{QDBioEq}\label{eq:QDBioEq}
\end{align}

\paragraph{Seed Demand}
\begin{align}
  QDSeed_{c,cty}
  &= \overline{qdSeed0}_{c,cty}
     \cdot
     \frac{\overline{pop}_{cty}}{\overline{pop0}_{cty}}
     \nonumber\\
  &\quad \cdot
     \prod_{c2}\!
     \left(
       \frac{\bigl(1-\overline{cse}_{c2,cty}\bigr)\cdot PC_{c2,cty}}
            {\bigl(1-\overline{cse0}_{c2,cty}\bigr)\cdot \overline{pc0}_{c2,cty}}
     \right)^{elaP_{c,c2,cty}}
  \tag{QDSeedEq}\label{eq:QDSeedEq}
\end{align}

\paragraph{Tourist Consumption Demand}
\begin{align}
  QDTour_{c,cty}
  &= \overline{qdTour0}_{c,cty}
     \cdot
     \prod_{c2}\!
     \left(
       \frac{\bigl(1-\overline{cse}_{c2,cty}\bigr)\cdot PC_{c2,cty}}
            {\bigl(1-\overline{cse0}_{c2,cty}\bigr)\cdot \overline{pc0}_{c2,cty}}
     \right)^{elaP_{c,c2,cty}}
  \tag{QDTourEq}\label{eq:QDTourEq}
\end{align}

\paragraph{Loss and Waste}
\begin{align}
  QDLoss_{c,cty}
  &= QD_{c,cty} \cdot \overline{lossRate}_{c,cty}
  \tag{QDLossEq}\label{eq:QDLossEq}\\[.3em]
  QDWaste_{c,cty}
  &= QDFood_{c,cty} \cdot \overline{wasteRate}_{c,cty}
  \tag{QDWasteEq}\label{eq:QDWasteEq}
\end{align}

\paragraph{Other Demand}
\begin{align}
  QDOth_{c,cty}
  &= \overline{qdOth0}_{c,cty}
     \cdot
     \prod_{c2}\!
     \left(
       \frac{\bigl(1-\overline{cse}_{c2,cty}\bigr)\cdot PC_{c2,cty}}
            {\bigl(1-\overline{cse0}_{c2,cty}\bigr)\cdot \overline{pc0}_{c2,cty}}
     \right)^{elaP_{c,c2,cty}}
  \tag{QDOthEq}\label{eq:QDOthEq}
\end{align}


\subsection{Production Block}

The production block describes the equations determining the production quantities of each food commodity, the input of intermediate goods and production factors, and the producer prices.  
The production functions are represented by Leontief-type functions, where each good is produced through the combination of specific input factors.

\subsubsection{Annual Driver Calculations}

Crop yields, livestock system shares, and feed input coefficients are calculated before each annual or sub-step equilibrium solve. They are therefore time-varying parameters rather than MCP equilibrium variables:

\begin{align*}
  \overline{yld}_{a\_cro,bas,cty,wat}
  &= \overline{yld0}_{a\_cro,bas,cty,wat}
     \cdot \overline{yldTech}_{a\_cro,bas,cty,wat}
     \\
  &\quad
     \cdot \overline{yldChg}_{a\_cro,bas,cty,wat}
     \cdot \overline{yldPlum}_{a\_cro,bas,cty},\\
  \overline{sysShr}_{a\_liv,bas,cty,cz,\mathrm{LG}}
  &= \frac{1}{1+\delta^{sys}_{a\_liv,bas,cty,cz,\mathrm{LG}}
     \cdot \overline{gdpPc}_{cty}^{-\rho^{sys}_{a\_liv,\mathrm{LG}}}},\\
  \overline{sysShr}_{a\_liv,bas,cty,cz,\mathrm{MX}}
  &= 1-\overline{sysShr}_{a\_liv,bas,cty,cz,\mathrm{LG}},\\
  \overline{feedCoef}_{a\_liv,c\_fed,bas,cty,cz,sys}
  &= \overline{feedReq}_{a\_liv,bas,cty,cz,sys}
     \cdot \overline{feedConcShr}_{a\_liv,bas,cty,cz,sys}
     \\
  &\quad
     \cdot \overline{feedGrainShr}_{a\_liv,c\_fed,bas,cty,cz,sys}.
\end{align*}

The optional \(yldPlum\) term is one when AIM-PLUM feedback is not used.

\paragraph{Domestic Production}
\begin{align}
  QS_{a,cty}
  &= \sum_{bas} QSBas_{a,bas,cty}
  \tag{QSAggEq}\label{eq:QSAggEq}
\end{align}

\subsubsection{Crop Production}

Primary crops are differentiated by production technologies (irrigated and rainfed).
Yield improvements due to technological progress and climate impacts are considered for crops.

\paragraph{Crop Production by Technology}
\begin{align}
  QSWat_{a\_cro,bas,cty,wat}
  &= QSBas_{a\_cro,bas,cty} \cdot watShr_{a\_cro,bas,cty,wat}
  \tag{QSWatEq}\label{eq:QSWatEq}
\end{align}

\paragraph{Producer Price of Crops}
\begin{align}
  PP_{a\_cro,cty}
  &= \Bigl(
      PL_{a\_cro,bas,cty}
      \cdot
      \sum_{wat}
      \frac{
        watShr_{a\_cro,bas,cty,wat}
      }{
        \overline{yld}_{a\_cro,bas,cty,wat}
        \cdot
        \overline{croInt}_{a\_cro,bas,cty,wat}
      }
      \nonumber\\[-0.3em]
  &\quad
      +\; COBas_{a\_cro,bas,cty}
     \Bigl)
     \,/\, (1 + MuBasMin_{a\_cro,bas,cty})
  \tag{CroZeroProfitEq}\label{eq:CroZeroProfitEq}
\end{align}

\paragraph{Land Input for Crop Production}
\begin{align}
  QL_{a\_cro,bas,cty}
  &= QSBas_{a\_cro,bas,cty}
     \cdot
     \sum_{wat}
     \frac{
       watShr_{a\_cro,bas,cty,wat}
     }{
       \overline{yld}_{a\_cro,bas,cty,wat}
       \cdot
       \overline{croInt}_{a\_cro,bas,cty,wat}
     }
  \tag{CroLandEq}\label{eq:CroLandEq}
\end{align}

\paragraph{Other Production Factor Price (Crops)}
\begin{align}
  COBas_{a\_cro,bas,cty}
  &= \overline{coBas0}_{a\_cro,bas,cty}
     \cdot
     \left(
       \frac{QSBas_{a\_cro,bas,cty}}
            {\overline{qsBasPrev}_{a\_cro,bas,cty}}
     \right)^{elaCO_{a\_cro,cty}}
  \tag{COCroEq}\label{eq:COCroEq}
\end{align}


\subsubsection{Livestock Production}

The livestock production block describes equations for determining livestock output, production systems, feed requirements, and associated prices.  
Livestock products are distinguished by climate zones and production systems (pasture-based or mixed).  
Feed use and production efficiency depend on feed composition, feed requirements, and animal productivity.

\paragraph{Livestock Production by System}
\begin{align}
  QSSys_{a\_liv,bas,cty,cz,sys}
  &= QSBas_{a\_liv,bas,cty}
     \cdot czShr_{a\_liv,bas,cty,cz}
     \cdot \overline{sysShr}_{a\_liv,bas,cty,cz,sys}
  \tag{QSSysEq}\label{eq:QSSysEq}
\end{align}

\paragraph{Producer Price of Livestock}
\begin{align}
  PP_{a\_liv,cty}
  &= \Bigl(
       PL_{a\_liv,bas,cty}
       / YLiv_{a\_liv,bas,cty}
       \nonumber\\[-0.3em]
  &\quad
       + feedCostCal_{a\_liv,cty}
         \cdot
         \sum_{c\_fed}
         (1 - \overline{cse}_{c\_fed,cty})
         \cdot
         PC_{c\_fed,cty}
         \nonumber\\[-0.3em]
  &\quad
         \cdot
         \sum_{cz,sys}
         \Bigl(
           czShr_{a\_liv,bas,cty,cz}
           \cdot
           \overline{sysShr}_{a\_liv,bas,cty,cz,sys}
           \cdot
           \overline{feedCoef}_{a\_liv,c\_fed,bas,cty,cz,sys}
         \Bigr)
         \nonumber\\[-0.3em]
  &\quad
       + COBas_{a\_liv,bas,cty}
     \Bigr)
     / (1 + MuBasMin_{a\_liv,bas,cty})
  \tag{LivZeroProfitEq}\label{eq:LivZeroProfitEq}
\end{align}

\paragraph{Grass Input Calculation}
\begin{align}
  QGrass_{a\_liv,bas,cty}
  &=
     \frac{
       QSBas_{a\_liv,bas,cty}
     }{
       \overline{animalProdRate}_{a\_liv,cty}
     }
     \nonumber\\[-0.3em]
  &\quad
     \cdot
     \sum_{cz,sys}
     \Bigl(
       czShr_{a\_liv,bas,cty,cz}
       \cdot
       \overline{sysShr}_{a\_liv,bas,cty,cz,sys}
       \nonumber\\[-0.3em]
  &\quad
       \cdot
       \overline{feedReq}_{a\_liv,bas,cty,cz,sys}
       \cdot
       (1 - \overline{feedConcShr}_{a\_liv,bas,cty,cz,sys})
     \Bigr)
  \tag{QGrassEq}\label{eq:QGrassEq}
\end{align}

\paragraph{Available Grassland Area}
\begin{align}
  QGrass_{a\_liv,bas,cty}
  &=
     QL_{a\_liv,bas,cty}
     \cdot
     \overline{yGrass}_{bas,cty}
  \tag{GrassBalEq}\label{eq:GrassBalEq}
\end{align}


\paragraph{Livestock Yield}
\begin{align}
  YLiv_{a\_liv,bas,cty}
  &= \frac{
       QSBas_{a\_liv,bas,cty}
     }{
       QL_{a\_liv,bas,cty}
     }
  \tag{YLivEq}\label{eq:YLivEq}
\end{align}

\paragraph{Feed Input Calculation}
\begin{align}
  QFeed_{a\_liv,c\_fed,bas,cty}
  &= \frac{
       QSBas_{a\_liv,bas,cty}
     }{
       \overline{animalProdRate}_{a\_liv,cty}
     }
     \nonumber\\[-0.3em]
  &\quad
     \cdot
     \sum_{cz,sys}
     \bigl(
       czShr_{a\_liv,bas,cty,cz}
       \cdot \overline{sysShr}_{a\_liv,bas,cty,cz,sys}
       \nonumber\\[-0.3em]
  &\quad
       \cdot \overline{feedCoef}_{a\_liv,c\_fed,bas,cty,cz,sys}
     \bigr)
  \tag{QFeedEq}\label{eq:QFeedEq}
\end{align}

\paragraph{Other Production Factor Price (Livestock)}
\begin{align}
  COBas_{a\_liv,bas,cty}
  &= \overline{coBas0}_{a\_liv,bas,cty}
     \cdot
     \left(
       \frac{
         QSBas_{a\_liv,bas,cty}
       }{
         \overline{qsBasPrev}_{a\_liv,bas,cty}
       }
     \right)^{elaCO_{a\_liv,cty}}
  \tag{COLivEq}\label{eq:COLivEq}
\end{align}

\paragraph{Livestock Production Quantity}
\begin{align}
  QSBas_{a\_liv,bas,cty}
  &= \overline{animalWt}_{a\_liv,cty}
     \cdot
     (
       NAnimal_{a\_liv,bas,cty}
       - \overline{nAnimalRes}_{a\_liv,bas,cty}
     )
     \nonumber\\[-0.3em]
  &\quad
     \cdot
     \overline{animalProdRate}_{a\_liv,cty}
  \tag{NAnimalEq}\label{eq:NAnimalEq}
\end{align}

\subsubsection{Processed Goods produciton}

This section describes the production equations for processed agricultural products.  
Processed products are produced using intermediate inputs and other production factors.

\paragraph{Producer Price of Processed Products}
\begin{align}
  PP_{a\_pro,cty}
  &= intCostCal_{a\_pro,cty}
     \cdot
     \sum_{c\_int}
     \Bigl(
       (1 - \overline{cse}_{c\_int,cty})
       \cdot
       PC_{c\_int,cty}
       \cdot
       intCoef_{a\_pro,c\_int,cty}
     \Bigr)
     \nonumber\\[-0.3em]
  &\quad
     + COCty_{a\_pro,cty}
  \tag{ProZeroProfitEq}\label{eq:ProZeroProfitEq}
\end{align}

\paragraph{Intermediate Input Demand for Processed Products}
\begin{align}
  QInt_{a\_pro,c\_int,cty}
  &= QS_{a\_pro,cty}
     \cdot
     intCoef_{a\_pro,c\_int,cty}
  \tag{QIntEq}\label{eq:QIntEq}
\end{align}

\paragraph{Other Production Factor Price (Processed Products)}
\begin{align}
  COCty_{a\_pro,cty}
  &= \overline{coCty0}_{a\_pro,cty}
     \cdot
     \left(
       \frac{
         QS_{a\_pro,cty}
       }{
         \overline{qsPrev}_{a\_pro,cty}
       }
     \right)^{elaCO_{a\_pro,cty}}
  \tag{COProEq}\label{eq:COProEq}
\end{align}

\subsubsection{Energy Crop Production}
Energy crop production is determined from regional totals provided exogenously by the AIM-Hub model~\cite{fujimoriSSP3AIMImplementation2017}.


\paragraph{Regional Energy Crop Supply Balance}
\begin{align}
  \overline{qdEne}_{reg}
  &= \sum_{bas,cty \in reg}
     QSBas_{\mathrm{ecr},bas,cty}
  \tag{EneBalEq}\label{eq:EneBalEq}
\end{align}

\paragraph{Land Input for Energy Crops}
\begin{align}
  QL_{\mathrm{ecr},bas,cty}
  &= \frac{
       QSBas_{\mathrm{ecr},bas,cty}
     }{
       \overline{yld}_{\mathrm{ecr},bas,cty,\mathrm{Rainfed}}
     }
  \tag{EneLandEq}\label{eq:EneLandEq}
\end{align}

\paragraph{Land Price for Energy Crops}
\begin{align}
  PL_{\mathrm{ecr},bas,cty}
  &= PLEne_{reg}
  \tag{PLEneEq}\label{eq:PLEneEq}
\end{align}


\subsection{Price Block}

This subsection describes the equations in the price block.  
The price block represents the relationships among various prices in the model.  
Prices are linked through fixed coefficients so that the price used in the demand and supply functions is consistently determined.

\paragraph{Consumer and Producer Prices}
\begin{align}
  (1 + \overline{mmj}_{a,cty})
  \cdot PP_{a,cty}
  + CEmis_{a,cty}
  + CAbt_{a,cty}
  &= PC_{a,cty}
  \tag{PCNtrdEq}\label{eq:PCNtrdEq}
\end{align}

\paragraph{Consumer and Border Prices}
\begin{align}
  PC_{c,cty}
  &= (1 + \overline{pse}_{c,cty})
     \cdot PB_{c,cty}
     \cdot (1 + pbRes_{c,cty})
  \tag{PCTrdEq}\label{eq:PCTrdEq}
\end{align}

\paragraph{Import Price}
\begin{align}
  PM_{c,cty}
  &= PW_{c}
     \cdot exr_{cty}
     \cdot
     (1 + \overline{tm}_{c,cty} + MuImpMin_{c,cty})
     \nonumber\\[-0.3em]
  &\quad
     \cdot (1 + \overline{mmm}_{c,cty})
     \cdot (1 + pmRes_{c,cty})
  \tag{PMEq}\label{eq:PMEq}
\end{align}

\paragraph{Export Price}
\begin{align}
  PE_{c,cty}
  \cdot (1 + \overline{te}_{c,cty})
  \cdot (1 + \overline{mme}_{c,cty})
  &= PW_{c}
     \cdot (1 + peRes_{c,cty})
     \cdot exr_{cty}
  \tag{PEEq}\label{eq:PEEq}
\end{align}


\subsection{Land Use Block}

The land-use block allocates available land among different uses through a Logit function.  
Non-pasture vegetation is allocated among cropland, forest, other natural land, and afforested land. Protected areas are imposed as lower bounds for the corresponding land classes, and the agro-ecological suitability constraint applies to cropland.  
Urban land is fixed exogenously. In the default configuration, the LUH2-consistent base-year urban area is held constant over future years; as an optional extension, SSP-specific urban projections from the SEDAC urban dataset can be used to update urban land over time.

\paragraph{Vegetation Area}
\begin{align}
  QLVeg_{bas,cty}
  &= 
     \overline{qlTot}_{bas,cty}
     - \overline{qlOth}_{bas,cty}
     - \overline{qlUrb}_{bas,cty}
  \tag{QLVegBalEq}\label{eq:QLVegBalEq}
\end{align}

\paragraph{Pasture Area Allocation}
\begin{align}
  QLPas_{bas,cty}
  &=
    QLVeg_{bas,cty}
    \nonumber\\[-0.3em]
  &\quad\cdot
    \frac{
      \delta^{\text{pas}}_{bas,cty}
      \cdot
      PLPas_{bas,cty}^{\,\rho^{\text{pas}}_{bas,cty}}
    }{
      \delta^{\text{pas}}_{bas,cty}
      \cdot
      PLPas_{bas,cty}^{\,\rho^{\text{pas}}_{bas,cty}}
      +
      \delta^{\text{pasn}}_{bas,cty}
      \cdot
      PLNpas_{bas,cty}^{\,\rho^{\text{pasn}}_{bas,cty}}
    }
  \tag{PasLogitEq}\label{eq:PasLogitEq}
\end{align}

\paragraph{Grazed and Ungrazed Pasture Allocation}
\begin{align}
  QLPasGrz_{bas,cty}
  &=
    QLPas_{bas,cty}
    \cdot
    \frac{
      \delta^{\text{pasgrz}}_{bas,cty}
      \cdot
      PLPasGrz_{bas,cty}^{\,\rho^{\text{pasgrz}}_{bas,cty}}
    }{
      \delta^{\text{pasgrz}}_{bas,cty}
      \cdot
      PLPasGrz_{bas,cty}^{\,\rho^{\text{pasgrz}}_{bas,cty}}
      +
      \delta^{\text{pasung}}_{bas,cty}
      \cdot
      PLPasUng_{bas,cty}^{\,\rho^{\text{pasung}}_{bas,cty}}
    }
  \tag{PasGrzLogitEq}\label{eq:PasGrzLogitEq}
\end{align}

\begin{align}
  QLPasUng_{bas,cty}
  &=
    QLPas_{bas,cty}
    \cdot
    \frac{
      \delta^{\text{pasung}}_{bas,cty}
      \cdot
      PLPasUng_{bas,cty}^{\,\rho^{\text{pasung}}_{bas,cty}}
    }{
      \delta^{\text{pasgrz}}_{bas,cty}
      \cdot
      PLPasGrz_{bas,cty}^{\,\rho^{\text{pasgrz}}_{bas,cty}}
      +
      \delta^{\text{pasung}}_{bas,cty}
      \cdot
      PLPasUng_{bas,cty}^{\,\rho^{\text{pasung}}_{bas,cty}}
    }
  \tag{PasUngLogitEq}\label{eq:PasUngLogitEq}
\end{align}


\paragraph{Non-pasture Grassland Area}
\begin{align}
  QLNpas_{bas,cty}
  &=
    QLVeg_{bas,cty}
    \nonumber\\[-0.3em]
  &\quad\cdot
    \frac{
      \delta^{\text{pasn}}_{bas,cty}
      \, PLNpas_{bas,cty}^{\,\rho^{\text{pasn}}_{bas,cty}}
    }{
      \delta^{\text{pas}}_{bas,cty}
      \, PLPas_{bas,cty}^{\,\rho^{\text{pas}}_{bas,cty}}
      +
      \delta^{\text{pasn}}_{bas,cty}
      \, PLNpas_{bas,cty}^{\,\rho^{\text{pasn}}_{bas,cty}}
    }
  \tag{NpasLogitEq}\label{eq:NpasLogitEq}
\end{align}

\paragraph{Cropland Area}
\begin{multline}
  QLCro_{bas,cty} = QLNpas_{bas,cty}\\
  \cdot
  \frac{
    \delta^{\text{cro}}_{bas,cty}\, PLCro_{bas,cty}^{\,\rho^{\text{cro}}_{bas,cty}}
  }{
    \delta^{\text{cro}}_{bas,cty}\, PLCro_{bas,cty}^{\,\rho^{\text{cro}}_{bas,cty}}
    + \delta^{\text{frs}}_{bas,cty}\, PLFrs_{bas,cty}^{\,\rho^{\text{frs}}_{bas,cty}}
    + \delta^{\text{gra}}_{bas,cty}\, PLNat_{bas,cty}^{\,\rho^{\text{gra}}_{bas,cty}}
    + \delta^{\text{afr}}_{bas,cty}\, PLAfr_{bas,cty}^{\,\rho^{\text{afr}}_{bas,cty}}
  }
  \tag{CroLogitEq}\label{eq:CroLogitEq}
\end{multline}

\paragraph{Forest Area}
\begin{multline}
  QLFrs_{bas,cty} = QLNpas_{bas,cty}\\
  \cdot
  \frac{
    \delta^{\text{frs}}_{bas,cty}\, PLFrs_{bas,cty}^{\,\rho^{\text{frs}}_{bas,cty}}
  }{
    \delta^{\text{cro}}_{bas,cty}\, PLCro_{bas,cty}^{\,\rho^{\text{cro}}_{bas,cty}}
    + \delta^{\text{frs}}_{bas,cty}\, PLFrs_{bas,cty}^{\,\rho^{\text{frs}}_{bas,cty}}
    + \delta^{\text{gra}}_{bas,cty}\, PLNat_{bas,cty}^{\,\rho^{\text{gra}}_{bas,cty}}
    + \delta^{\text{afr}}_{bas,cty}\, PLAfr_{bas,cty}^{\,\rho^{\text{afr}}_{bas,cty}}
  }
  \tag{FrsLogitEq}\label{eq:FrsLogitEq}
\end{multline}

\paragraph{Primary and Secondary Forest Allocation}
\begin{align}
  QLPrimf_{bas,cty}
  &= QLFrs_{bas,cty}
     \cdot
     \frac{
       \delta^{\text{primf}}_{bas,cty}\, PLPrimf_{bas,cty}^{\,\rho^{\text{primf}}_{bas,cty}}
     }{
       \delta^{\text{primf}}_{bas,cty}\, PLPrimf_{bas,cty}^{\,\rho^{\text{primf}}_{bas,cty}}
       +
       \delta^{\text{secdf}}_{bas,cty}\, PLSecdf_{bas,cty}^{\,\rho^{\text{secdf}}_{bas,cty}}
     }
  \tag{PrimfLogitEq}\label{eq:PrimfLogitEq}\\[0.4em]
  QLSecdf_{bas,cty}
  &= QLFrs_{bas,cty}
     \cdot
     \frac{
       \delta^{\text{secdf}}_{bas,cty}\, PLSecdf_{bas,cty}^{\,\rho^{\text{secdf}}_{bas,cty}}
     }{
       \delta^{\text{primf}}_{bas,cty}\, PLPrimf_{bas,cty}^{\,\rho^{\text{primf}}_{bas,cty}}
       +
       \delta^{\text{secdf}}_{bas,cty}\, PLSecdf_{bas,cty}^{\,\rho^{\text{secdf}}_{bas,cty}}
     }
  \tag{SecdfLogitEq}\label{eq:SecdfLogitEq}
\end{align}

\paragraph{Other Natural Land}
\begin{multline}
  QLNat_{bas,cty} = QLNpas_{bas,cty}\\
  \cdot
  \frac{
    \delta^{\text{gra}}_{bas,cty}\, PLNat_{bas,cty}^{\,\rho^{\text{gra}}_{bas,cty}}
  }{
    \delta^{\text{cro}}_{bas,cty}\, PLCro_{bas,cty}^{\,\rho^{\text{cro}}_{bas,cty}}
    + \delta^{\text{frs}}_{bas,cty}\, PLFrs_{bas,cty}^{\,\rho^{\text{frs}}_{bas,cty}}
    + \delta^{\text{gra}}_{bas,cty}\, PLNat_{bas,cty}^{\,\rho^{\text{gra}}_{bas,cty}}

  }

\end{multline}

\paragraph{Afforested Land}
\begin{multline}
  QLAfr_{bas,cty} = QLNpas_{bas,cty}\\
  \cdot
  \frac{
    \delta^{\text{afr}}_{bas,cty}\, PLAfr_{bas,cty}^{\,\rho^{\text{afr}}_{bas,cty}}
  }{
    \delta^{\text{cro}}_{bas,cty}\, PLCro_{bas,cty}^{\,\rho^{\text{cro}}_{bas,cty}}
    + \delta^{\text{frs}}_{bas,cty}\, PLFrs_{bas,cty}^{\,\rho^{\text{frs}}_{bas,cty}}
    + \delta^{\text{gra}}_{bas,cty}\, PLNat_{bas,cty}^{\,\rho^{\text{gra}}_{bas,cty}}
    + \delta^{\text{afr}}_{bas,cty}\, PLAfr_{bas,cty}^{\,\rho^{\text{afr}}_{bas,cty}}
  }
  \tag{AfrLogitEq}\label{eq:AfrLogitEq}
\end{multline}

\paragraph{Harvested Cropland Area}
\begin{multline}
  QLCroCul_{bas,cty} = QLCro_{bas,cty}\\
  \cdot
  \frac{
    \delta^{\text{croclt}}_{bas,cty}\, PLCroCul_{bas,cty}^{\,\rho^{\text{croclt}}_{bas,cty}}
  }{
    \delta^{\text{croclt}}_{bas,cty}\, PLCroCul_{bas,cty}^{\,\rho^{\text{croclt}}_{bas,cty}}
    + \delta^{\text{crofal}}_{bas,cty}\, PLCroFal_{bas,cty}^{\,\rho^{\text{crofal}}_{bas,cty}}
  }
  \tag{CroCulLogitEq}\label{eq:CroCulLogitEq}
\end{multline}

\paragraph{Fallow Cropland Area}
\begin{multline}
  QLCroFal_{bas,cty} = QLCro_{bas,cty}\\
  \cdot
  \frac{
    \delta^{\text{crofal}}_{bas,cty}\, PLCroFal_{bas,cty}^{\,\rho^{\text{crofal}}_{bas,cty}}
  }{
    \delta^{\text{croclt}}_{bas,cty}\, PLCroCul_{bas,cty}^{\,\rho^{\text{croclt}}_{bas,cty}}
    + \delta^{\text{crofal}}_{bas,cty}\, PLCroFal_{bas,cty}^{\,\rho^{\text{crofal}}_{bas,cty}}
  }
  \tag{CroFalLogitEq}\label{eq:CroFalLogitEq}
\end{multline}


\paragraph{Cropland Allocation}
\begin{align}
  QL_{a\_cro,bas,cty}
  &= QLCroCul_{bas,cty}
     \cdot
     \frac{
       \delta_{a\_cro,bas,cty}\cdot PL_{a\_cro,bas,cty}^{\,\rho_{a\_cro,bas,cty}}
     }{
       \sum_{a\_cro}\delta_{a\_cro,bas,cty}\cdot PL_{a\_cro,bas,cty}^{\,\rho_{a\_cro,bas,cty}}
     }
  \tag{CroDistEq}\label{eq:CroDistEq}
\end{align}

\paragraph{Pasture Allocation}
\begin{align}
  QL_{a\_liv,bas,cty}
  &= QLPasGrz_{bas,cty}
     \cdot
     \frac{
       \delta_{a\_liv,bas,cty}\cdot PL_{a\_liv,bas,cty}^{\,\rho_{a\_liv,bas,cty}}
     }{
       \sum_{a\_liv}\delta_{a\_liv,bas,cty}\cdot PL_{a\_liv,bas,cty}^{\,\rho_{a\_liv,bas,cty}}
     }
  \tag{LivDistEq}\label{eq:LivDistEq}
\end{align}


\paragraph{Land Prices}
The price drivers for other natural land and afforestation include their land-carbon value before the associated complementarity wedges are applied.
\begin{equation}
  PLCroCul_{bas,cty}\cdot QLCroCul_{bas,cty}
  = \sum_{a\_cro} PL_{a\_cro,bas,cty}\cdot QL_{a\_cro,bas,cty}
  \tag{PLCroCulEq}\label{eq:PLCroCulEq}
\end{equation}

\vspace{0.3\baselineskip}

\begin{multline}
  PLCro_{bas,cty}\cdot QLCro_{bas,cty}
  = \bigl(
       PLCroCul_{bas,cty}\cdot QLCroCul_{bas,cty}
       \\[-0.3em]
  \quad
       + PLCroFal_{bas,cty}\cdot QLCroFal_{bas,cty}
     \bigr)
  \quad / (1 + MuCroCap_{bas,cty})
  \tag{PLCroEq}\label{eq:PLCroEq}
\end{multline}

\vspace{0.3\baselineskip}

\begin{multline}
  PLNpas_{bas,cty}^{\,\rho^{\text{pasn}}_{bas,cty}}
  \cdot \overline{qlNpas0}_{bas,cty}
  = \overline{plNpas0}_{bas,cty}^{\,\rho^{\text{pasn}}_{bas,cty}}
  \\
  \cdot \left(
    \delta^{\text{cro}} PLCro^{\rho^{\text{cro}}}
    + \delta^{\text{frs}} PLFrs^{\rho^{\text{frs}}}
    + \delta^{\text{gra}} PLNat^{\rho^{\text{gra}}}
    + \delta^{\text{afr}} PLAfr^{\rho^{\text{afr}}}
  \right)_{bas,cty}
  \tag{PLNpasEq}\label{eq:PLNpasEq}
\end{multline}

\vspace{0.3\baselineskip}

\begin{equation}
  PLPasGrz_{bas,cty}\cdot QLPasGrz_{bas,cty}
  = \sum_{a\_liv} PL_{a\_liv,bas,cty}\cdot QL_{a\_liv,bas,cty}
  \tag{PLPasGrzEq}\label{eq:PLPasGrzEq}
\end{equation}

\vspace{0.3\baselineskip}

\begin{multline}
  PLPas_{bas,cty}\cdot QLPas_{bas,cty}
  =
  \\[-0.3em]
  \quad 
  \left(
    PLPasGrz_{bas,cty}\cdot QLPasGrz_{bas,cty}
    +
    PLPasUng_{bas,cty}\cdot QLPasUng_{bas,cty}
  \right)
  \\[-0.3em]
  \quad \cdot (1 + MuPasProt_{bas,cty})
  \tag{PLPasEq}\label{eq:PLPasEq}
\end{multline}

\vspace{0.3\baselineskip}

\begin{multline}
  PLFrs_{bas,cty}^{\,\rho^{\text{frs}}_{bas,cty}}
  \cdot \overline{qlFrs0}_{bas,cty}
  = \overline{plFrs0}_{bas,cty}^{\,\rho^{\text{frs}}_{bas,cty}}
  \\
  \cdot \left(
    \delta^{\text{primf}} PLPrimf^{\rho^{\text{primf}}}
    + \delta^{\text{secdf}} PLSecdf^{\rho^{\text{secdf}}}
  \right)_{bas,cty}
  \tag{PLFrsEq}\label{eq:PLFrsEq}
\end{multline}

\begin{align}
  PLPrimf_{bas,cty}
  &= \Biggl[
    \overline{plPrimf0}_{bas,cty}
    + \max\!\Biggl(
        0,\,
        \overline{carbonStock}_{bas,cty}\cdot\frac{44}{12}
        \nonumber\\[-0.3em]
  &\qquad\qquad
        \cdot \overline{pCarLand}_{bas,cty}
        \cdot dcffrs_{cty}
        - afrCost_{cty}
      \Biggr)
  \Biggr]
  \nonumber\\[-0.3em]
  &\quad \cdot
  \frac{1 + MuPrimfProt_{bas,cty}}
       {1 + MuPrimfCap_{bas,cty}}
  \tag{PLPrimfEq}\label{eq:PLPrimfEq}
\end{align}

\begin{align}
  PLSecdf_{bas,cty}
  &= \Biggl[
    \overline{plSecdf0}_{bas,cty}
    + \max\!\Biggl(
        0,\,
        \overline{carbonStock}_{bas,cty}\cdot\frac{44}{12}
        \nonumber\\[-0.3em]
  &\qquad\qquad
        \cdot \overline{pCarLand}_{bas,cty}
        \cdot dcffrs_{cty}
        - afrCost_{cty}
      \Biggr)
  \Biggr]
  \nonumber\\[-0.3em]
  &\quad \cdot
  \frac{1 + MuSecdfProt_{bas,cty}}
       {1 + MuSecdfCap_{bas,cty}}
  \tag{PLSecdfEq}\label{eq:PLSecdfEq}
\end{align}

\begin{align}
  PLNat_{bas,cty}
  &=
  \overline{plNatNet}_{bas,cty}
  \cdot (1+MuNatProt_{bas,cty})
  \tag{PLNatEq}\label{eq:PLNatEq}
\end{align}

\paragraph{Afforestation Price and Annual Limit}

  PLAfr_{bas,cty}

  QLAfr_{bas,cty}

  QLSecdf_{bas,cty}

\subsection{Market Block}

This subsection describes the equations in the market block, where consistency between food demand and supply is ensured.  
Trade of food commodities is represented as a complementarity problem.  
When the import price is higher than the border price, the import quantity becomes zero.  
Once the import price decreases to equal the border price, imports begin.  
Similarly, when the border price is higher than the export price, exports become zero, and once the export price rises to match the border price, exports take place.

\paragraph{Domestic Trade Balance}
\begin{align}
  NT_{c,cty}
  &= QS_{c,cty} - QD_{c,cty} - \overline{qStock}_{c,cty}
  \tag{ComBalEq}\label{eq:ComBalEq}\\[0.6em]
  NT_{c,cty}
  &= QE_{c,cty} - QM_{c,cty}
  \tag{NTEq}\label{eq:NTEq}
\end{align}

\paragraph{International Trade Balance}
\begin{align}
  \sum_{cty} NT_{c,cty} - \overline{qRes}_{c} = 0
  \tag{WldBalEq}\label{eq:WldBalEq}
\end{align}

\paragraph{Import and Export Conditions}
\begin{align}
  PM_{c,cty} &\ge PB_{c,cty}
  \quad \perp \quad
  QM_{c,cty} \ge 0
  \tag{ImpParityEq}\label{eq:ImpParityEq}\\[0.4em]
  PB_{c,cty} &\ge PE_{c,cty}
  \quad \perp \quad
  QE_{c,cty} \ge 0
  \tag{ExpParityEq}\label{eq:ExpParityEq}
\end{align}



\subsection{GHG Emission Block}

This subsection describes the equations in the GHG emission block.  
GHG emissions are calculated based on the production quantities of each food commodity.  
When carbon prices are considered, the amount of mitigation technology adoption and its cost are computed from the MAC (Marginal Abatement Cost) curve.  
Emissions from the land-use sector are calculated according to each land area.

\paragraph{GHG Emissions}
\begin{align}
  GHG_{a,src,gas,cty}
  &= QS_{a,cty}
     \cdot
     \overline{ef}_{a,src,gas,cty}
     \cdot
     (1 - RRed_{a,src,gas,cty})
  \tag{GHGAgEq}\label{eq:GHGAgEq}
\end{align}

\paragraph{Land-Use Emissions}
\begin{multline}
  GHGLU_{src,gas,cty}
  =
  \sum_{bas}\Bigl(
    \overline{efLUCro}_{src,gas,cty}\cdot QLCro_{bas,cty}
    \\[-0.3em]
    + \overline{efLUFrs}_{src,gas,cty}\cdot QLFrs_{bas,cty}
    \\[-0.3em]
    - \overline{carbonStock}_{bas,cty}\cdot\frac{44}{12}
      \cdot \min\!\Bigl\{
        0,\,
        QLFrs_{bas,cty}-QLFrs^{t-1}_{bas,cty}
      \Bigr\}
    \\[-0.3em]
    + \overline{efLUPas}_{src,gas,cty}\cdot QLPas_{bas,cty}
    + \overline{efLUNat}_{src,gas,cty}\cdot QLNat_{bas,cty}
  \Bigr)
  \tag{GHGLUEq}\label{eq:GHGLUEq}
\end{multline}
\begin{align}
  GHGLU_{"afr","CO_2",cty}
  &=
  - \sum_{bas}\sum_{age}
      qlAfrAge_{bas,cty,age}
      \cdot \overline{carbonSeq}_{bas,cty,age}
  \tag{AfrGHGEq}\label{eq:AfrGHGEq}
\end{align}

\paragraph{Emission Reduction Rate}
\begin{align}
  RRed_{a,src,gas,cty}
  &= \overline{mrp}_{src,gas,cty}
     \cdot
     \Bigl(
       1 -
       \bigl[
         1 / (1 + \overline{pCar}_{cty})
       \bigr]^{\sigma_{src,gas,cty}}
     \Bigr)
  \tag{RRedEq}\label{eq:RRedEq}
\end{align}

\paragraph{Abatement Cost}
\begin{align}
  CAbt_{a,cty}
  &= \sum_{src,gas}
     \overline{ef}_{a,src,gas,cty}
     \cdot
     \Biggl[
       \frac{\sigma_{src,gas,cty}}
            {(\sigma_{src,gas,cty}-1)}
       \cdot
       \overline{mrp}_{src,gas,cty}
       \nonumber\\[-0.3em]
  &\quad
       \cdot
       \Bigl(
         1 -
         \bigl(
           1 -
           RRed_{a,src,gas,cty} /
           \overline{mrp}_{src,gas,cty}
         \bigr)^{(1 - 1/\sigma_{src,gas,cty})}
       \Bigr)
       - RRed_{a,src,gas,cty}
     \Biggr]
  \tag{CAbtEq}\label{eq:CAbtEq}
\end{align}

\paragraph{Carbon Cost on Remaining Emissions}
\begin{align}
  CEmis_{a,cty}
  &= \sum_{src,gas}
     \overline{pCar}_{cty}
     \cdot
     \overline{ef}_{a,src,gas,cty}
     \cdot
     (1 - RRed_{a,src,gas,cty})
  \tag{CEmisEq}\label{eq:CEmisEq}
\end{align}


\subsection{Complementarity Conditions}

This subsection describes the complementarity conditions introduced to avoid numerical errors in the Logit function when production quantities in specific production units become zero.  
If the production of a unit becomes zero, its land-use area would also become zero, leading to instability.  
Therefore, a minimum positive value is imposed to ensure that production quantities do not fall to zero.

\paragraph{Minimum Production Share for Trade}
\begin{align}
  \frac{QS_{c,cty}}{QM_{c,cty}}
  &\ge 0.01
  \quad \perp \quad
  MuImpMin_{c,cty} \ge 0
  \tag{ImpMinEq}\label{eq:ImpMinEq}
\end{align}

\paragraph{Minimum Production Share for Supply}
\begin{align}
  \frac{QSBas_{a,bas,cty}}{QS_{a,cty}}
  &\ge 0.01
  \quad \perp \quad
  MuBasMin_{a,bas,cty} \ge 0
  \tag{QSBasMinEq}\label{eq:QSBasMinEq}
\end{align}
